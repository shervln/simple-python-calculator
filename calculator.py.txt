# ماشین حساب ساده با پایتون

def جمع(a, b):
    return a + b

def تفریق(a, b):
    return a - b

def ضرب(a, b):
    return a * b

def تقسیم(a, b):
    if b != 0:
        return a / b
    else:
        return "خطا: تقسیم بر صفر!"

print("ماشین حساب ساده")
print("عملیات: جمع (+), تفریق (-), ضرب (*), تقسیم (/)")

while True:
    عمل = input("عملیات را وارد کن (+, -, *, / یا 'خروج' برای تمام کردن): ")
    if عمل == "خروج":
        print("خدانگهدار!")
        break

    عدد1 = float(input("عدد اول: "))
    عدد2 = float(input("عدد دوم: "))

    if عمل == "+":
        print("نتیجه:", جمع(عدد1, عدد2))
    elif عمل == "-":
        print("نتیجه:", تفریق(عدد1, عدد2))
    elif عمل == "*":
        print("نتیجه:", ضرب(عدد1, عدد2))
    elif عمل == "/":
        print("نتیجه:", تقسیم(عدد1, عدد2))
    else:
        print("عملیات نامعتبر!")
